-- =====================================================================
-- 1. CREACIÓN DE TABLAS MAESTRAS Y TRANSACCIONALES
-- =====================================================================

-- Eliminar tablas si ya existen (orden inverso por dependencias de llaves foráneas)
DROP TABLE IF EXISTS reportes_estado CASCADE;
DROP TABLE IF EXISTS autorizaciones CASCADE;
DROP TABLE IF EXISTS vehiculos CASCADE;
DROP TABLE IF EXISTS conductores CASCADE;

-- Tabla de Conductores (30 registros)
CREATE TABLE conductores (
    id SERIAL PRIMARY KEY,
    rut VARCHAR(12) UNIQUE NOT NULL,
    nombre_completo VARCHAR(150) NOT NULL,
    grado_o_cargo VARCHAR(50) NOT NULL,
    unidad_destino VARCHAR(100) NOT NULL,
    estado VARCHAR(20) DEFAULT 'Activo' -- Activo, Suspendido, En Comisión
);

-- Tabla de Vehículos (15 registros)
CREATE TABLE vehiculos (
    id SERIAL PRIMARY KEY,
    patente VARCHAR(10) UNIQUE NOT NULL,
    tipo_vehiculo VARCHAR(50) NOT NULL, -- Táctico, SUV, Camión, Bus, Automóvil
    marca_modelo VARCHAR(100) NOT NULL,
    anio INT NOT NULL,
    kilometraje_actual INT DEFAULT 0,
    estado_operativo VARCHAR(30) DEFAULT 'Operativo' -- Operativo, Requiere Mantención, Crítico
);

-- Tabla de Autorizaciones (Relación N a N entre Conductores y Vehículos)
CREATE TABLE autorizaciones (
    id SERIAL PRIMARY KEY,
    patente VARCHAR(10) REFERENCES vehiculos(patente) ON UPDATE CASCADE ON DELETE CASCADE,
    rut_conductor VARCHAR(12) REFERENCES conductores(rut) ON UPDATE CASCADE ON DELETE CASCADE,
    fecha_asignacion TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    vigente BOOLEAN DEFAULT TRUE
);

-- Tabla de Reportes de Estado y Check-in (Bitácora del Conductor)
CREATE TABLE reportes_estado (
    id SERIAL PRIMARY KEY,
    patente VARCHAR(10) REFERENCES vehiculos(patente) ON UPDATE CASCADE ON DELETE CASCADE,
    rut_conductor VARCHAR(12) REFERENCES conductores(rut) ON UPDATE CASCADE ON DELETE CASCADE,
    fecha_reporte TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    kilometraje_salida INT NOT NULL,
    nivel_combustible INT CHECK (nivel_combustible BETWEEN 0 AND 100), -- Porcentaje 0-100
    presion_neumaticos BOOLEAN NOT NULL, -- True: OK, False: Baja Presión
    luces_operativas BOOLEAN NOT NULL, -- True: OK, False: Falla
    observaciones TEXT,
    foto_evidencia_url TEXT -- URL de Supabase Storage
);
